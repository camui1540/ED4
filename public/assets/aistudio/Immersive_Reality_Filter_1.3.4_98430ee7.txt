  class AudioEngine {
    constructor(src, callbacks, title = '', theme = CONFIG.DEFAULT_THEME) {
      this.src = src;
      this.title = title;
      this.theme = theme;
      this.callbacks = callbacks;
      this.audioCtx = null;
      this.audioEl = null;
      this.filterNode = null;
      this.gainNode = null;
      this.isInitialized = false;
      this.isPlaying = false;
      this.isMuffled = false;
      this.isTailFading = false;
      this.fadeTimerId = null;
      engines.add(this);
    }

    initAudioGraph() {
      if (this.isInitialized) return;

      const AudioCtx = window.AudioContext || window.webkitAudioContext;
      this.audioCtx = new AudioCtx();

      this.audioEl = new Audio();
      this.audioEl.crossOrigin = 'anonymous';
      this.audioEl.src = this.src;
      this.audioEl.loop = playMode === 'loop';
      this.audioEl.preload = 'metadata';

      this.audioEl.addEventListener('ended', () => {
        if (isLiveMode && liveRole === 'GM') {
          if (playMode === 'once') {
            this.pause(0);
            currentTrack = null;
            if (channel) {
              channel.send({
                type: 'broadcast',
                event: 'bgm',
                payload: { url: this.src, action: 'pause', fade: isLongFade },
              });
            }
          } else if (playMode === 'next') {
            this.pause(0);
            const playlist = [...engines].filter((e) => !remotePlayer || e !== remotePlayer.engine);
            if (!playlist.length) return;
            const curIdx =
              currentTrack && playlist.includes(currentTrack.eng)
                ? playlist.indexOf(currentTrack.eng)
                : playlist.findIndex((e) => e.src === this.src);
            const nextEng = playlist[(curIdx + 1) % playlist.length];
            if (nextEng) {
              currentTrack = {
                url: nextEng.src,
                title: nextEng.title,
                theme: nextEng.theme,
                eng: nextEng,
              };
              handleAudioAction(nextEng.src, 'play', nextEng.title, nextEng.theme);
              if (channel) {
                channel.send({
                  type: 'broadcast',
                  event: 'bgm',
                  payload: {
                    url: nextEng.src,
                    action: 'play',
                    title: nextEng.title,
                    theme: nextEng.theme,
                    fade: isLongFade,
                  },
                });
              }
            }
          }
        } else if (!isLiveMode) {
          this.pause(0);
        }
      });

      const sourceNode = this.audioCtx.createMediaElementSource(this.audioEl);

      this.filterNode = this.audioCtx.createBiquadFilter();
      this.filterNode.type = 'lowpass';
      this.filterNode.frequency.setValueAtTime(
        this.isMuffled ? CONFIG.FILTER_FREQ_MUFFLED : CONFIG.FILTER_FREQ_NORMAL,
        this.audioCtx.currentTime
      );
      this.filterNode.Q.setValueAtTime(0.707, this.audioCtx.currentTime);

      this.gainNode = this.audioCtx.createGain();
      this.gainNode.gain.setValueAtTime(0.0001, this.audioCtx.currentTime);

      sourceNode.connect(this.filterNode);
      this.filterNode.connect(this.gainNode);
      this.gainNode.connect(this.audioCtx.destination);

      const updateTime = () => {
        const currentTime = this.audioEl.currentTime || 0;
        const duration = this.audioEl.duration || 0;
        const remaining = duration - currentTime;

        if (
          isLongFade &&
          this.isPlaying &&
          duration > CONFIG.LONG_FADE_DURATION * 2 &&
          remaining <= CONFIG.LONG_FADE_DURATION &&
          remaining > 0.1
        ) {
          if (!this.isTailFading) {
            this.isTailFading = true;
            rampParam(this.gainNode.gain, 0.0001, remaining, this.audioCtx.currentTime);
          }
        } else if (this.isTailFading && (!isLongFade || remaining > CONFIG.LONG_FADE_DURATION)) {
          this.isTailFading = false;
          if (this.isPlaying) {
            rampParam(this.gainNode.gain, 1.0, getFadeSec(), this.audioCtx.currentTime);
          }
        }

        this.callbacks.onTimeUpdate({
          currentTime,
          duration,
          progress: duration > 0 ? currentTime / duration : 0,
        });
      };

      this.audioEl.addEventListener('timeupdate', updateTime);
      this.audioEl.addEventListener('loadedmetadata', updateTime);
      this.audioEl.addEventListener('error', () => this.callbacks.onError());

      this.isInitialized = true;
    }

    async play(fadeDuration = getFadeSec()) {
      this.initAudioGraph();
      if (this.audioCtx.state === 'suspended') await this.audioCtx.resume();

      for (const engine of engines) {
        if (engine !== this && engine.isPlaying) engine.pause();
      }

      clearTimeout(this.fadeTimerId);
      this.isTailFading = false;
      this.gainNode.gain.setValueAtTime(0.0001, this.audioCtx.currentTime);
      rampParam(this.gainNode.gain, 1.0, fadeDuration, this.audioCtx.currentTime);

      this.isPlaying = true;
      this.callbacks.onStateChange(this);
      await this.audioEl.play();
    }

    pause(fadeDuration = getFadeSec()) {
      if (!this.isPlaying) return;

      clearTimeout(this.fadeTimerId);
      this.isPlaying = false;
      this.isTailFading = false;
      this.callbacks.onStateChange(this);

      if (fadeDuration <= 0) {
        this.gainNode.gain.cancelScheduledValues(this.audioCtx.currentTime);
        this.gainNode.gain.setValueAtTime(0.0001, this.audioCtx.currentTime);
        this.audioEl.pause();
      } else {
        rampParam(this.gainNode.gain, 0.0001, fadeDuration, this.audioCtx.currentTime);
        this.fadeTimerId = setTimeout(() => this.audioEl.pause(), fadeDuration * 1000);
      }
    }

    toggleMuffled() {
      this.isMuffled = !this.isMuffled;
      if (this.isInitialized) {
        const target = this.isMuffled ? CONFIG.FILTER_FREQ_MUFFLED : CONFIG.FILTER_FREQ_NORMAL;
        rampParam(
          this.filterNode.frequency,
          target,
          CONFIG.FILTER_TRANSITION_TIME,
          this.audioCtx.currentTime,
          true
        );
      }
      this.callbacks.onStateChange(this);
    }

    seekByRatio(ratio) {
      if (!this.audioEl || !this.audioEl.duration) return;
      this.audioEl.currentTime = clamp(ratio, 0, 1) * this.audioEl.duration;
    }
  }

    static loadRemoteCss(cssUrl) {
      if (remoteCssLoaded) return;
      const targetUrl = `${cssUrl}${cssUrl.includes('?') ? '&' : '?'}t=${Date.now()}`;

      GM_xmlhttpRequest({
        method: 'GET',
        url: targetUrl,
        onload: (res) => {
          if (res.status >= 200 && res.status < 300 && res.responseText) {
            remoteCssLoaded = true;
            const raw = res.responseText.replace(/<\/?style[^>]*>/gi, '');
            const imports = [];
            const cleanCss = raw.replace(
              /@import\s+(?:url\()?['"]?([^'"\)]+)['"]?\)?\s*;/gi,
              (_, url) => {
                imports.push(url);
                return '';
              }
            );
            const safeCss = cleanCss.replace(/(?:^|\})\s*(?::root|\*|body|\.forum-[^{]+)\s*\{[^}]*\}\s*/gi, '}').replace(/^}\s*/, '');
document.getElementById(CONFIG.REMOTE_STYLE_ID).textContent = safeCss;
            for (const fontUrl of imports) {
              const link = document.createElement('link');
              link.rel = 'stylesheet';
              link.href = fontUrl;
              document.head.appendChild(link);
            }
          }
        },
      });

          liveSetup.classList.toggle('is-open');
        }
      });

   connectBtn.addEventListener('click', () => {
     const room = roomInput.value.trim();
     if (!room) return;
     liveRole = roleSelect.value;

     if (liveRole === 'PL') {
       new (window.AudioContext || window.webkitAudioContext)().resume();
       document.createElement('audio').play().catch(() => {});
     }

     const broadcastCurrent = () => {
       if (isLiveMode && liveRole === 'GM' && currentTrack && channel) {
         const curEng = [...engines].find((e) => e.src === currentTrack.url && e.isPlaying);
         if (curEng) {
           channel.send({
             type: 'broadcast',
             event: 'bgm',
             payload: {
               url: currentTrack.url,
               action: 'play',
               title: currentTrack.title,
               theme: currentTrack.theme,
               currentTime: curEng.audioEl ? curEng.audioEl.currentTime : 0,
               fade: isLongFade,
             },
           });
         }
       }
     };

     supabaseClient =
       supabaseClient || window.supabase.createClient(CONFIG.SUPABASE_URL, CONFIG.SUPABASE_KEY);
     channel = supabaseClient.channel(`trpg-bgm-${room}`, {
       config: { broadcast: { self: false } },
     });
     channel
       .on('presence', { event: 'sync' }, () => {
         hasPeers = Object.values(channel.presenceState()).flat().length > 1;
         broadcastCurrent();
       })
       .on('presence', { event: 'join' }, () => {
         hasPeers = true;
         broadcastCurrent();
       })
       .on('broadcast', { event: 'sync_req' }, () => {
         hasPeers = true;
         broadcastCurrent();
       })
       .on('broadcast', { event: 'bgm_fade' }, ({ payload }) => {
         if (liveRole === 'PL') isLongFade = Boolean(payload.fade);
       })
       .on('broadcast', { event: 'bgm' }, ({ payload }) => {
         hasPeers = true;
         if (liveRole === 'PL') {
           if (payload.fade !== undefined) isLongFade = Boolean(payload.fade);
           handleAudioAction(payload.url, payload.action, payload.title, payload.theme, payload.currentTime);
         }
       })
       .subscribe((status) => {
         if (status === 'SUBSCRIBED') {
           channel.track({ role: liveRole });
           if (liveRole === 'PL') {
             channel.send({ type: 'broadcast', event: 'sync_req' });
           }
         }
       });

     isLiveMode = true;
     liveSetup.classList.remove('is-open');
     syncAllLiveBtns();
     syncAllModeSelects();
   });

    static attachFloatAndDragBehavior(container, placeholder, floatBtn, dragHandle) {
      let isFloating = false;
      let isDragging = false;
      let dragOffsetX = 0;
      let dragOffsetY = 0;
      let savedPos = null;


      const setPos = (left, top) => {
        const rect = container.getBoundingClientRect();
        const x = clamp(left, 8, Math.max(8, window.innerWidth - rect.width - 8));
        const y = clamp(top, 8, Math.max(8, window.innerHeight - rect.height - 8));
        container.style.left = `${x}px`;
        container.style.top = `${y}px`;
        return { left: x, top: y };
      };

      const applyDefaultOrSavedPos = () => {
        const rect = container.getBoundingClientRect();
        const pos = savedPos || {
          left: window.innerWidth - rect.width - 20,
          top: window.innerHeight - rect.height - 20,
        };
        setPos(pos.left, pos.top);
      };

      floatBtn.addEventListener('click', () => {
        isFloating = !isFloating;
        container.dataset.floating = String(isFloating);
        container.classList.toggle('is-floating', isFloating);
        placeholder.classList.toggle('is-active', isFloating);
        syncBtn(floatBtn, isFloating, 'Dock', 'Float');

        if (isFloating) {
          document.body.appendChild(container);
          applyDefaultOrSavedPos();
        } else {
          container.style.left = '';
          container.style.top = '';
          placeholder.parentNode.insertBefore(container, placeholder.nextSibling);
        }
      });

      dragHandle.addEventListener('pointerdown', (e) => {
        if (!isFloating || e.target.closest('button, input, select')) return;
        isDragging = true;
        const rect = container.getBoundingClientRect();
        dragOffsetX = e.clientX - rect.left;
        dragOffsetY = e.clientY - rect.top;
        container.classList.add('is-dragging');
        dragHandle.setPointerCapture(e.pointerId);
        e.preventDefault();
      });

      dragHandle.addEventListener('pointermove', (e) => {
        if (!isDragging) return;
        savedPos = setPos(e.clientX - dragOffsetX, e.clientY - dragOffsetY);
      });

      const stopDrag = (e) => {
        if (!isDragging) return;
        isDragging = false;
        container.classList.remove('is-dragging');
        if (dragHandle.hasPointerCapture(e.pointerId)) {
          dragHandle.releasePointerCapture(e.pointerId);
        }
      };

      dragHandle.addEventListener('pointerup', stopDrag);
      dragHandle.addEventListener('pointercancel', stopDrag);
      window.addEventListener('resize', () => isFloating && applyDefaultOrSavedPos());
    }

    static formatTime(current, duration) {
      const fmt = (sec) => {
        const s = Math.max(0, Math.floor(sec || 0));
        return `${String(Math.floor(s / 60)).padStart(2, '0')}:${String(s % 60).padStart(2, '0')}`;
      };
      return duration > 0 ? `${fmt(current)} / ${fmt(duration)}` : fmt(current);
    }
  }

  class DOMObserver {
    static parseAudioAnchor(anchor) {
      const rawHref = anchor.getAttribute('href');
      if (!rawHref || !/^[a-zA-Z0-9+.-]+:|^\//.test(rawHref)) return null;

      const parsedUrl = new URL(rawHref, window.location.href);
      const cleanPathname = parsedUrl.pathname.replace(/\/+$/, '');
      const cleanPath = cleanPathname.toLowerCase();
      const isSupportedAudio = CONFIG.AUDIO_EXTENSIONS.some((ext) => cleanPath.endsWith(ext));
      if (!isSupportedAudio) return null;

      const theme = parsedUrl.hash.slice(1).trim() || CONFIG.DEFAULT_THEME;
      const title =
        (anchor.innerText || anchor.textContent).trim() ||
        decodeURIComponent(cleanPathname.split('/').pop() || 'Unknown BGM');

      return {
        url: `${parsedUrl.origin}${cleanPathname}${parsedUrl.search}`,
        theme,
        title,
      };
    }

    scanAndReplace(root = document) {
      const anchors =
        root.tagName === 'A'
          ? [root]
          : root.querySelectorAll(`a[href]:not([${CONFIG.PROCESSED_ATTR}])`);

      for (const anchor of anchors) {
        const meta = DOMObserver.parseAudioAnchor(anchor);
        if (!meta) continue;

        anchor.setAttribute(CONFIG.PROCESSED_ATTR, 'true');
        const { container, placeholder } = UIBuilder.createPlayerComponent(meta);
        anchor.parentNode.insertBefore(placeholder, anchor);
        anchor.parentNode.insertBefore(container, anchor);
        anchor.parentNode.removeChild(anchor);
      }
    }

    start() {
      UIBuilder.injectStructuralStyles();
      this.scanAndReplace(document);

      new MutationObserver((mutations) => {
        for (const { addedNodes } of mutations) {
          for (const node of addedNodes) {
            if (
              node.nodeType === 1 &&
              !node.classList.contains('trpg-bgm-container') &&
              !node.classList.contains('trpg-bgm-inline-placeholder')
            ) {
              this.scanAndReplace(node);
            }
          }
        }
      }).observe(document.body, { childList: true, subtree: true });
    }
  }

  new DOMObserver().start();
